      // Generate timestamp for cache busting
      const pageLoadTimestamp = Date.now();

      // Function to mark content as loaded and hide loader
      function markContentLoaded() {
        document.body.classList.add("content-loaded");
        setTimeout(() => {
          const loader = document.getElementById("loading-overlay");
          if (loader) {
            loader.style.display = "none";
          }
        }, 500);

      // Handle various loading events
      window.addEventListener("DOMContentLoaded", () => {
        // Check if CSS is loaded
        const sheets = document.styleSheets;
        if (sheets.length > 1) {
          markContentLoaded();
        } else {
          // Fallback if CSS takes longer
          setTimeout(markContentLoaded, 1000);
        }

        // Add page load timestamp as a data attribute to help with debugging
        document.documentElement.setAttribute(
          "data-page-loaded",
          pageLoadTimestamp.toString()
        );
      });

      // Fallback if nothing else triggers
      window.addEventListener("load", markContentLoaded);

      // Clear any cached data on page load
      function clearCachedData() {
        try {
          // Try to clear application cache
          if (window.applicationCache && window.applicationCache.swapCache) {
            window.applicationCache.swapCache();
          }

          // Set a reload flag in sessionStorage
          sessionStorage.setItem("page-reloaded", "true");
          sessionStorage.setItem(
            "last-reload-time",
            pageLoadTimestamp.toString()
          );
        } catch (e) {
          console.log("Cache clearing error:", e);
        }
      }
